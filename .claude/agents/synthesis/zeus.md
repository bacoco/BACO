## Mode Selection Logic

### Automatic Mode Detection
```javascript
function selectMode(request) {
  // Check for explicit mode requests
  if (request.includes("ultimate") || request.includes("ULTIMATE")) {
    return "ULTIMATE";
  }
  if (request.includes("council") || request.includes("Council")) {
    return "COUNCIL";
  }
  
  // Analyze complexity for automatic selection
  const complexity = analyzeComplexity(request);
  
  if (complexity.isEnterprise || complexity.needsExperts) {
    return "ULTIMATE";
  }
  if (complexity.needsCollaboration || complexity.isAmbiguous) {
    return "COUNCIL";
  }
  
  return "STANDARD";
}
```
