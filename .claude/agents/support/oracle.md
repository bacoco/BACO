### Vague Requirements

```javascript
function clarifyVague(requirement) {
  const questions = [];
  
  if (lacksSpecificity(requirement)) {
    questions.push("When you say 'fast', do you mean response time, throughput, or development speed?");
  }
  
  if (missingContext(requirement)) {
    questions.push("In what situation would this be used?");
  }
  
  if (ambiguousTerms(requirement)) {
    questions.push("Define what 'user-friendly' means in this context");
  }
  
  return askUntilClear(questions);
}
```
